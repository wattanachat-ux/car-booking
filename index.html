// ============================================================ // code.gs — Guts IOC Car Booking Backend // ============================================================ const SPREADSHEET_ID = "1r770uF-s89a4hBtMc99ak3DnnMwvSAmEo8G5uaF_OWQ"; const SHEET_NAME = "Sheet1"; const ADMIN_PASSWORDS = { "11223344": "จิรศักดิ์", "12121212": "จุติมา" }; const KEYMAN_PASSWORDS = { "33334444": "วัฒนะชาติ", "55556666": "โกเมน", "77778888": "พิชชากร" }; // ============================================================ function doGet(e) { const action = e.parameter.action; const ss = SpreadsheetApp.openById(SPREADSHEET_ID); const sheet = ss.getSheetByName(SHEET_NAME); if (!action) return getAll(sheet); if (action === "login") return loginCheck(e, ADMIN_PASSWORDS); if (action === "loginKeyman") return loginCheck(e, KEYMAN_PASSWORDS); if (action === "save") return saveBooking(e, sheet); if (action === "updateStatus") return updateStatus(e, sheet); if (action === "updateKeyStatus") return updateKeyStatus(e, sheet); if (action === "delete") return deleteRow(e, sheet); return ContentService.createTextOutput("Unknown action"); } // ── โหลดข้อมูล ✅ แปลงเวลาเป็น "HH:MM" string ก่อนส่ง แก้ปัญหาเวลาผิด ── function getAll(sheet) { const data = sheet.getDataRange().getValues(); const headers = data[0].map(h => h.toString().toLowerCase().trim()); const json = data.slice(1).map(row => { let obj = {}; headers.forEach((h, i) => { let val = row[i]; // ถ้า header เป็น start หรือ end และค่าเป็น Date object → แปลงเป็น "HH:MM" if ((h === "start" || h === "end") && val instanceof Date) { const hh = String(val.getHours()).padStart(2, '0'); const mm = String(val.getMinutes()).padStart(2, '0'); val = hh + ":" + mm; } obj[h] = val; }); return obj; }); return jsonOutput(json); } function loginCheck(e, passwordMap) { const name = passwordMap[e.parameter.pwd || ""]; return jsonOutput(name ? { ok: true, name: name } : { ok: false }); } function saveBooking(e, sheet) { const p = e.parameter; sheet.appendRow([ p.id || "", p.date || "", p.car || "", p.project || "", p.driver || "", p.p1 || "", p.p2 || "", p.start || "", p.end || "", "รออนุมัติ", "", "", "", "" ]); return ContentService.createTextOutput("Saved"); } function updateStatus(e, sheet) { const id = String(e.parameter.id || ""); const data = sheet.getDataRange().getValues(); for (let i = 1; i < data.length; i++) { if (String(data[i][0]) === id) { sheet.getRange(i + 1, 10).setValue(e.parameter.status || ""); sheet.getRange(i + 1, 11).setValue(e.parameter.admin || ""); break; } } return ContentService.createTextOutput("Updated"); } function updateKeyStatus(e, sheet) { const id = String(e.parameter.id || ""); const data = sheet.getDataRange().getValues(); for (let i = 1; i < data.length; i++) { if (String(data[i][0]) === id) { sheet.getRange(i + 1, 13).setValue(e.parameter.keystatus || ""); sheet.getRange(i + 1, 12).setValue(e.parameter.keyman || ""); if (e.parameter.keystatus === "คืนแล้ว") { sheet.getRange(i + 1, 14).setValue(new Date().toISOString()); } break; } } return ContentService.createTextOutput("Updated"); } function deleteRow(e, sheet) { const id = String(e.parameter.id || ""); const data = sheet.getDataRange().getValues(); for (let i = 1; i < data.length; i++) { if (String(data[i][0]) === id) { sheet.deleteRow(i + 1); break; } } return ContentService.createTextOutput("Deleted"); } function jsonOutput(obj) { return ContentService.createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); }
